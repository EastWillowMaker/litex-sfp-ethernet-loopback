"""Measure the authorized two-NIC FPGA bridge, then restore host settings."""
import hashlib
import json
import os
from pathlib import Path
import sys
import time
project = Path(__file__).resolve().parents[2]
sys.path.insert(0, str(project))
os.chdir(project)
from sfp_loopback import bridge_host as b
phase = sys.argv[1]
if phase not in ('baseline', 'benchmark', 'qualify'):
    raise ValueError('Unknown benchmark phase')
root = project/'artifacts/bridge500'
run_dir = root/(time.strftime('%Y%m%d-%H%M%S')+'-'+phase)
run_dir.mkdir(parents=True, exist_ok=False)
csr = project/('build/sfp_bridge/csr.csv' if phase == 'baseline' else 'build/sfp_bridge500/csr.csv')
runner = b.SystemRunner()
interfaces = {'enx98fc84e0c4e8':'98:fc:84:e0:c4:e8','enx00e04c5828c8':'00:e0:4c:58:28:c8'}
state = None
initial = {}
receipt = {'phase':phase,'target_mbps':500,'passed':False,'restoration':{},'source_sha256':hashlib.sha256((project/'sfp_loopback/bridge_host.py').read_bytes()).hexdigest()}
state_path = run_dir/'host-state.json'
profiles = ('Wired connection 1','Wired connection 2')
def profile_settings():
    return {name: runner.run(['nmcli','-g','connection.uuid,connection.interface-name,802-3-ethernet.mac-address,ipv4.method,ipv6.method','connection','show',name]).stdout for name in profiles}
profiles_before = profile_settings()
receipt['profiles_before'] = profiles_before
# NetworkManager's generated wired profiles live in /run and can disappear
# with their device. Preserve their keyfiles while the isolated test owns it.
profile_files = {}
for line in runner.run(['nmcli','-t','-f','UUID,FILENAME','connection','show']).stdout.splitlines():
    uuid, filename = line.split(':',1)
    if uuid not in {value.splitlines()[0] for value in profiles_before.values()}: continue
    path = Path(filename)
    if path.parent != Path('/run/NetworkManager/system-connections'):
        raise RuntimeError('Unexpected generated wired-profile location')
    profile_files[path] = (path.read_bytes(), path.stat().st_mode & 0o777)
receipt['profile_file_hashes_before'] = {str(p):hashlib.sha256(data).hexdigest() for p,(data,mode) in profile_files.items()}
try:
    inventory = b.inspect_host(runner, iperf3_path=str(project/'artifacts/bridge/iperf3'))
    for name, mac in interfaces.items():
        item = inventory['physical_interfaces'][name]
        if item['mac'] != mac or not item['carrier'] or item['speed_mbps'] != 1000:
            raise RuntimeError(f'{name}: expected identity and 1000 Mbps carrier required')
        if name in inventory['default_route_interfaces'] or name in inventory['ssh_route_interfaces']:
            raise RuntimeError(f'{name}: management interface cannot be moved')
        initial[name] = b._nm_state(runner,name)
    (run_dir/'initial-management.json').write_text(json.dumps(initial,indent=2)+'\n')
    for name in interfaces:
        runner.run(['nmcli','device','set',name,'managed','no'],timeout=10)
    time.sleep(1)
    state = b.setup_pair('enx98fc84e0c4e8','enx00e04c5828c8',state_path=state_path,confirm_isolation=True,runner=runner)
    before = b._read_hardware_status(b.DEFAULT_PORT,csr)
    (run_dir/'before.json').write_text(json.dumps(before,indent=2)+'\n')
    if not all(port['link_up'] and not port['loss_of_signal'] for port in before['links'].values()):
        raise RuntimeError('Both FPGA PCS links must be up before throughput measurement')
    if phase == 'qualify':
        result = b.run_bridge_test(state,output_dir=run_dir/'qualification',runner=runner,csr_csv=csr,
            iperf3_path=str(project/'artifacts/bridge/iperf3'),iperf_duration=30)
        runs = result['performance']['runs']
        receipt['correctness_status'] = result['overall']
        correctness = result['overall'] == 'PASS'
    else:
        runs = [b._run_iperf(runner,state,source,dest,3,run_dir,str(project/'artifacts/bridge/iperf3'))
                for source,dest in [('left','right'),('right','left')]]
        correctness = True
        receipt['correctness_status'] = 'NOT RUN: diagnostic throughput only'
    after = b._read_hardware_status(b.DEFAULT_PORT,csr)
    (run_dir/'after.json').write_text(json.dumps(after,indent=2)+'\n')
    receipt['runs'] = runs
    receipt['throughput_passed'] = len(runs)==2 and all(x['status']=='PASS' and x['receiver_bits_per_second']>=500e6 for x in runs)
    receipt['passed'] = correctness and receipt['throughput_passed']
    receipt['directional_counter_deltas'] = {direction:{key:(value-before['forwarding'][direction][key])%2**32 for key,value in counters.items()} for direction,counters in after['forwarding'].items()}
except BaseException as error:
    receipt['error'] = f'{type(error).__name__}: {error}'
finally:
    if state is not None:
        try:
            restored = b.restore_state(state,state_path=state_path,runner=runner)
            receipt['restoration']['namespaces'] = {'restored':restored['restored'],'errors':restored['restore_errors']}
        except BaseException as error:
            receipt['restoration']['namespaces'] = {'restored':False,'error':str(error)}
    # Restore volatile profiles before re-enabling management, preventing NM
    # from allocating replacement UUIDs/names when the devices reappear.
    receipt['profile_reload_errors'] = []
    for path,(data,mode) in profile_files.items():
        try:
            path.write_bytes(data)
            path.chmod(mode)
            runner.run(['nmcli','connection','load',str(path)],timeout=10)
        except BaseException as error:
            receipt['profile_reload_errors'].append({'error':str(error),'stderr':getattr(error,'stderr',None)})
    # Restore this run's state; historical profile labels are not NIC identities.
    for name,saved in initial.items():
        if saved['managed'] is not None:
            try:
                runner.run(['nmcli','device','set',name,'managed','yes' if saved['managed'] else 'no'],timeout=10)
            except BaseException as error:
                receipt['restoration'][name] = {'passed':False,'error':str(error), 'stderr':getattr(error,'stderr',None)}
    time.sleep(1)
    for name,saved in initial.items():
        try:
            if saved['managed'] and saved.get('connection'):
                runner.run(['nmcli','--wait','0','connection','up','id',saved['connection'],'ifname',name],timeout=10)
                time.sleep(1)
            current = b._nm_state(runner,name)
            # With no active saved profile, normal NM autoactivation may start DHCP.
            receipt['restoration'][name] = {'saved':saved,'current':current,'passed':current['managed']==saved['managed'] and (not saved.get('connection') or current['connection']==saved['connection'])}
        except BaseException as error:
            receipt['restoration'][name] = {'passed':False,'error':str(error),'stderr':getattr(error,'stderr',None)}
    # The wrapper reactivates NM after standalone restoration. Verify final
    # observable state as well as preserving all earlier restoration errors.
    final_checks = {}
    if state is not None:
        live_namespaces = {line.split()[0] for line in runner.run(['ip','netns','list']).stdout.splitlines() if line.split()}
        final_checks['owned_namespaces_absent'] = not live_namespaces.intersection(state['owned_namespaces'])
        for record in state['interfaces'].values():
            name = record['interface']
            links = json.loads(runner.run(['ip','-j','address','show','dev',name]).stdout)
            routes = []
            for family in ([],['-6']):
                routes.extend({**item,'dev':name} for item in json.loads(runner.run(['ip','-j',*family,'route','show','table','all','dev',name]).stdout))
            link = links[0]
            final_checks[name] = {
                'identity':link['address'].lower()==record['mac'],
                'mtu':link['mtu']==record['mtu'],
                'up':('UP' in link['flags'])==record['was_up'],
                'test_address_absent':all(a.get('local')!=record['test_ipv4'] for a in link.get('addr_info',[])),
                'saved_addresses_present':all(any(a.get('local')==old.get('local') and a.get('prefixlen')==old.get('prefixlen') and bool(a.get('noprefixroute'))==bool(old.get('noprefixroute')) for a in link.get('addr_info',[])) for old in record['addresses']),
                'saved_routes_present':all(any(b._route_signature(a)==b._route_signature(old) for a in routes) for old in record['routes'] if b._is_link_local_route(old)),
            }
    receipt['final_host_verification'] = final_checks
    receipt['restore_passed'] = bool(final_checks.get('owned_namespaces_absent')) and all(all(final_checks.get(name,{'missing':False}).values()) and receipt['restoration'].get(name,{}).get('passed') for name in interfaces)
    receipt['profiles_after'] = profile_settings()
    receipt['profiles_unchanged'] = receipt['profiles_before'] == receipt['profiles_after']
    receipt['restore_passed'] = receipt['restore_passed'] and receipt['profiles_unchanged'] and not receipt['profile_reload_errors']
    receipt['passed'] = receipt['passed'] and receipt['restore_passed']
    (run_dir/'result.json').write_text(json.dumps(receipt,indent=2)+'\n')
    (root/'latest-result.json').write_text(json.dumps({'directory':str(run_dir),'result':receipt},indent=2)+'\n')
    uid = int(os.environ.get('SUDO_UID','1000'))
    for p in run_dir.rglob('*'):
        if p.is_file() and not p.is_symlink():os.chown(p,uid,-1)
    os.chown(root/'latest-result.json',uid,-1)
    print(json.dumps(receipt,indent=2),flush=True)
sys.exit(0 if receipt['passed'] else 1)
